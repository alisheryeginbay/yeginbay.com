{{- /* Makes each heading that has an id a link to itself, with a § the CSS shows at the right edge of the column as a hint. A heading that already holds a link or a sidenote can't go inside another link, so the § alone is its link instead, and so it is for one inside a , where a click opens the details. Org puts the heading text on its own line, so whitespace around it goes, or it would indent the text. Takes a dict with "html", and optionally "margin" set to true to always use the § link. Returns the HTML with the links added. */ -}} {{- $out := .html -}} {{- $whole := not .margin -}} {{- range $m := findRESubmatch `(?s)\s*(.*?)\s*` $out -}} {{- $level := index $m 1 -}} {{- $id := index $m 2 -}} {{- $text := index $m 3 -}} {{- $new := "" -}} {{- if and $whole (not (findRE `<(a|label|input)[\s>]` $text 1)) -}} {{- $new = printf `§%s` $level $id $id $text $level -}} {{- else -}} {{- $new = printf `§%s` $level $id $id $text $level -}} {{- end -}} {{- $out = replace $out (index $m 0) $new 1 -}} {{- end -}} {{- return ($out | safeHTML) -}}
