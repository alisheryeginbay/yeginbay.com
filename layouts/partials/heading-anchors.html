{{- /* Makes each heading that has an id a link to itself, with a § the CSS hangs in the margin to its left as a hint, and puts a button after the heading text that copies the link (assets/js/heading-copy.ts). A heading that already holds a link or a sidenote can't go inside another link, so the § alone is its link instead, and so it is for one inside a , where a click opens the details. Org puts the heading text on its own line, so whitespace around it goes, or it would indent the text. Takes a dict with "html", and optionally "unlinked" set to true to leave the heading text unlinked. Returns the HTML with the links and buttons added. */ -}} {{- $copy := `` -}} {{- $out := .html -}} {{- $unlinked := .unlinked -}} {{- range $m := findRESubmatch `(?s)\s*(.*?)\s*` $out -}} {{- $level := index $m 1 -}} {{- $id := index $m 2 -}} {{- $text := index $m 3 -}} {{- if or $unlinked (findRE `<(a|label|input)[\s>]` $text 1) -}} {{- $text = printf `§%s` $id $text -}} {{- else -}} {{- $text = printf `§%s` $id $text -}} {{- end -}} {{- $new := printf `%s%s` $level $id $text $copy $level -}} {{- $out = replace $out (index $m 0) $new 1 -}} {{- end -}} {{- return ($out | safeHTML) -}}
