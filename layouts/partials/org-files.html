{{- /* Returns the paths of all .org files under the given directory. */ -}} {{- $files := slice -}} {{- range os.ReadDir . -}} {{- $path := path.Join $ .Name -}} {{- if .IsDir -}} {{- $files = $files | append (partial "org-files.html" $path) -}} {{- else if strings.HasSuffix .Name ".org" -}} {{- $files = $files | append $path -}} {{- end -}} {{- end -}} {{- return $files -}}
